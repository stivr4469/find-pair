Команда `mkdir -p` используется для создания директории или директорий и всех промежуточных директорий, если они не существуют. Например, если вы хотите создать директорию `/path/to/directory`, но директории `/path` и `/path/to` еще не существуют, команда `mkdir -p` автоматически создаст их.

Вот пример использования команды `mkdir -p`:

```sh
mkdir -p /path/to/directory
```

Эта команда создаст директорию `/path/to/directory` и все промежуточные директории,

########################################################################################################################################server {
    listen 80;  # Указывает на порт, на котором сервер будет прослушивать входящие HTTP-запросы.

    server_name localhost;  # Указывает на имя сервера, которое будет обрабатывать входящие запросы. В этом случае, сервер будет обрабатывать запросы для хоста "localhost".

    index index.php;  # Указывает на файлы-индексы, которые будут использоваться, если в URL не указано имя файла. Здесь указано, что при отсутствии конкретного файла будет использоваться index.php.

    root /var/www/public_html;  # Указывает на корневую директорию, в которой будут искаться файлы для обработки запросов.

    location ~ \.php$ {  # Определяет блок настроек для обработки запросов к PHP-скриптам. Здесь используется регулярное выражение, которое ищет строки, заканчивающиеся на .php.

        try_files $uri =404;  # Пытается найти файл в файловой системе, соответствующий запрошенному URI. Если файл не найден, вернет код ошибки 404.

        fastcgi_pass 172.17.0.2:9000;  # Указывает на адрес и порт FastCGI-сервера (например, PHP-FPM), который будет обрабатывать PHP-скрипты.

        fastcgi_index index.php;  # Указывает на файл-индекс, который будет использоваться для запросов к директории, если не указан конкретный файл.

        include fastcgi_params;  # Включает файл с общими параметрами для FastCGI-прокси.

        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;  # Устанавливает значение переменной SCRIPT_FILENAME, которая указывает на путь к запрашиваемому файлу PHP.
    }
}
